rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Jeder Shop hat seinen eigenen Bestellnummer-Zähler.
    match /counters/{customerId} {
      allow read, write: if true;
    }

    // Kunden dürfen nur neue Bestellungen anlegen.
    // Lesen, Status ändern und Löschen ist ausschließlich dem zentralen Admin erlaubt.
    match /orders/{orderId} {
      allow create: if request.resource.data.customerId is string
                    && request.resource.data.customerId.size() > 0;
      allow read, update, delete: if request.auth != null
        && request.auth.token.email == "textilien@proton.me";
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
